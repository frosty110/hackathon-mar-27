       ```yaml
       lint:

           - STANDARD
       breaking:

           - FILE
       ```

    3. Create `buf.gen.yaml` in project root. The `go_package_prefix` must match the Go module path:
       ```yaml
       version: v2
       managed:
         enabled: true
         override:
           - file_option: go_package_prefix
             value: github.com/blaisealbuquerque/pricing-radar/gen
       plugins:
         - local: [go, tool, protoc-gen-go]
           out: gen
           opt: paths=source_relative
         - local: [go, tool, protoc-gen-connect-go]
           out: gen
           opt:
             - paths=source_relative
             - simple
       inputs:
         - directory: proto
       ```
