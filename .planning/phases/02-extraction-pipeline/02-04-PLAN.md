**Step 3 — Rewrite RunScan with extract-normalize-store pipeline:**

The existing flow is: NewScanRun → FetchAll → for each page: StripBoilerplate → SaveSnapshot → FinishScanRun.

The new flow adds concurrent extraction between strip and store:

```go
func (h *PricingHandler) RunScan(
    ctx context.Context,
    req *pricingv1.RunScanRequest,
) (*pricingv1.RunScanResponse, error) {
    // 1. Create new scan run.
    scanRunID, err := h.db.NewScanRun(ctx)
    if err != nil {
        return nil, err
    }

    // 2. Fetch all target pages concurrently.
    rawPages := scraper.FetchAll(ctx, h.targets)

    // 3. Strip boilerplate from each page.
    type strippedPage struct {
        page    scraper.RawPage
        html    string
    }
    stripped := make([]strippedPage, len(rawPages))
    for i, page := range rawPages {
        s, err := scraper.StripBoilerplate(page.HTML)
        if err != nil {
            slog.Warn("StripBoilerplate failed", "competitor", page.Competitor, "error", err)
            s = page.HTML
        }
        stripped[i] = strippedPage{page: page, html: s}
    }

    // 4. Save snapshots and run extraction concurrently (errgroup).
    type pipelineResult struct {
        competitor string
        snapshotID int64
        data       *normalizer.RawPricingData
        fromCache  bool
        scrapedAt  string
        extractErr error
    }
    results := make([]pipelineResult, len(stripped))
    var g errgroup.Group
    for i, sp := range stripped {
        i, sp := i, sp
        g.Go(func() error {
            // 4a. Save snapshot (get id for later update).
            snapshotID, snapErr := h.db.SaveSnapshot(ctx, scanRunID, sp.page.Competitor, sp.html, sp.page.FromCache)
            if snapErr != nil {
                slog.Warn("SaveSnapshot failed", "competitor", sp.page.Competitor, "error", snapErr)
                // Continue — use snapshotID=0 to skip SaveExtractedPricing
            }

            // 4b. LLM extraction (non-fatal on failure).
            raw, extractErr := h.extractor.Extract(ctx, sp.page.Competitor, sp.html)
            if extractErr != nil {
                slog.Warn("extraction failed", "competitor", sp.page.Competitor, "error", extractErr)
                if snapshotID > 0 {
                    _ = h.db.SaveExtractionError(ctx, snapshotID, extractErr)
                }
                results[i] = pipelineResult{
                    competitor: sp.page.Competitor,
                    snapshotID: snapshotID,
                    fromCache:  sp.page.FromCache,
                    scrapedAt:  sp.page.FetchedAt.Format("2006-01-02T15:04:05Z07:00"),
                    extractErr: extractErr,
                }
                return nil
            }

            // 4c. Normalize (mutates raw in place, returns same pointer).
            normalized := normalizer.Normalize(raw)

            // 4d. Persist extracted + normalized fields.
            if snapshotID > 0 {
                if err := h.db.SaveExtractedPricing(ctx, snapshotID, normalized); err != nil {
                    slog.Warn("SaveExtractedPricing failed", "competitor", sp.page.Competitor, "error", err)
                }
            }

            results[i] = pipelineResult{
                competitor: sp.page.Competitor,
                snapshotID: snapshotID,
                data:       normalized,
                fromCache:  sp.page.FromCache,
                scrapedAt:  sp.page.FetchedAt.Format("2006-01-02T15:04:05Z07:00"),
            }
            return nil
        })
    }
    _ = g.Wait()

    // 5. Build response.
    resp := &pricingv1.RunScanResponse{ScanRunId: scanRunID}
    for _, r := range results {
        cr := &pricingv1.CompetitorResult{
            Competitor: r.competitor,
            FromCache:  r.fromCache,
            ScrapedAt:  r.scrapedAt,
        }
        if r.data != nil {
            cr.PricingModel             = r.data.PricingModel
            cr.NormalizedMonthlyCost    = r.data.NormalizedMonthlyCost
            cr.NormalizationAssumptions = r.data.NormalizationAssumptions
            cr.Confidence               = r.data.Confidence
            cr.PricingPublic            = r.data.PricingPublic
            cr.CannotNormalize          = r.data.CannotNormalize
        }
        if r.extractErr != nil {
            cr.ExtractionError = r.extractErr.Error()
        }
        resp.Results = append(resp.Results, cr)
    }

    // 6. Mark scan run as finished.
    if err := h.db.FinishScanRun(ctx, scanRunID); err != nil {
        slog.Warn("FinishScanRun failed", "scan_run_id", scanRunID, "error", err)
    }

    return resp, nil
}
```
