```go
func (e *Extractor) Extract(ctx context.Context, competitor, strippedHTML string) (*normalizer.RawPricingData, error) {
    // 1. Truncate stripped HTML to 12000 chars (prevents context_length_exceeded)
    html := truncate(strippedHTML, 12000)

    // 2. Build request payload
    payload := chatRequest{
        Model: e.cfg.TrueFoundryCheapModel,
        Messages: []chatMessage{
            {Role: "system", Content: systemPrompt},
            {Role: "user", Content: "Extract pricing data from this HTML for competitor: " + competitor + "\n\n" + html},
        },
        ResponseFormat: responseFormat{Type: "json_object"},
        Temperature:    0.0,
    }

    // 3. Marshal, POST, read response
    body, err := json.Marshal(payload)
    // ... http.NewRequestWithContext, set Content-Type and Authorization headers ...
    // ... decode chatResponse ...

    // 4. Check for API-level error in response body
    if chatResp.Error != nil {
        return nil, fmt.Errorf("TrueFoundry error for %s: %s", competitor, chatResp.Error.Message)
    }
    if len(chatResp.Choices) == 0 {
        return nil, fmt.Errorf("empty choices from TrueFoundry for %s", competitor)
    }
    content := chatResp.Choices[0].Message.Content

    // 5. Unmarshal into typed struct; on failure try stripping markdown fences
    var result normalizer.RawPricingData
    if err := json.Unmarshal([]byte(content), &result); err != nil {
        // Try stripping ```json ... ``` wrapper (Pitfall 1)
        stripped := stripMarkdownFences(content)
        if err2 := json.Unmarshal([]byte(stripped), &result); err2 != nil {
            slog.Error("malformed LLM JSON", "competitor", competitor, "preview", content[:min(100, len(content))], "error", err2)
            return nil, fmt.Errorf("malformed LLM JSON for %s: %w", competitor, err2)
        }
    }

    // 6. Post-process: fix confidence > 1.0 (Pitfall 2 — LLM returns percentage)
    if result.Confidence > 1.0 {
        result.Confidence = result.Confidence / 100.0
    }

    // 7. Ensure company and URL are set
    if result.Company == "" {
        result.Company = competitor
    }

    return &result, nil
}
```

**Helper functions:**
- `truncate(s string, n int) string` — returns s[:n] if len(s) > n, else s
- `stripMarkdownFences(s string) string` — trims leading "```json\n" or "```\n" and trailing "```" using strings.TrimPrefix/TrimSuffix + strings.TrimSpace
- `min(a, b int) int` — returns smaller of two ints
